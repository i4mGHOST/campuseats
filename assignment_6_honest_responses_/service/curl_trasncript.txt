# 1. Success Transcript: Happy Path (201 Created with Location Header & JSON Body)

$ curl -i -X POST http://localhost:5000/orders \
-H "Content-Type: application/json" \
-H "Accept: application/json" \
-H "Idempotency-Key: key_demo_001" \
-d '{
      "studentId": "usr_883920194",
      "paymentMethodId": "pm_card_visa_4242",
      "items": [
        {
          "itemId": "item_01",
          "name": "Veggie Supreme Burger",
          "quantity": 2,
          "unitPriceCents": 725
        }
      ]
    }'

Content-Type: application/json; charset=utf-8

Content-Length: 268

{
  "createdAt": "2026-10-13T10:15:30Z",

    {

# 3. Field-Level Validation Failure (422 Unprocessable Entity with errors[] array)

$ curl -i -X POST http://localhost:5000/orders \
-H "Content-Type: application/json" \
-d '{
      "studentId": "",
      "paymentMethodId": "pm_card_visa_4242",
      "items": [
        {
          "itemId": "item_01",
          "name": "Veggie Burger",
          "quantity": 0,
          "unitPriceCents": 725
        }
      ]
    }'

HTTP/1.1 422 UNPROCESSABLE ENTITY
Content-Type: application/problem+json; charset=utf-8

{
  "detail": "Multiple request fields failed validation checks.",
  "errors": [
    {
      "field": "studentId",
      "reason": "Field 'studentId' is required and must be a non-empty string."
    },
    {
      "field": "items[0].quantity",
      "reason": "Quantity must be an integer greater than or equal to 1."
    }
  ],
  "status": 422,
  "title": "Unprocessable Entity",
  "type": "https://api.campuseats.com/errors/validation-error"
}
