-H "Content-Type: application/json" \ 
-H "Idempotency-Key: key_demo_001" \ 
-d '{ 
		"studentId": "usr_883920194",
		"paymentMethodId": "pm_card_visa_4242", 
		"items": [ {
			"itemId": "item_01",

			"quantity": 2,

	}'

# 2. Idempotent Repeat (Same Idempotency-Key Returns Original Result) 

$ curl -i -X POST http://localhost:5000/orders \ 
-H "Content-Type: application/json" \ 
-H "Idempotency-Key: key_demo_001" \ 
-d 
	'{ 
		"studentId": "usr_883920194", 
		"paymentMethodId": "pm_card_visa_4242", 
		"items": [ {
					"itemId": "item_01", 
					"name": "Veggie Supreme Burger", 
					"quantity": 2, 
					"unitPriceCents": 725} ] }'

-H "Content-Type: application/json" \ 
-d '{"reason": "Student requested order cancellation."}'

Content-Type: application/json

{
